# SleepRadioPi boot config (Pi Zero 2 W). The boot partition is mounted
# read-only at /boot; edit this file on a PC if you need to change it.

# Cut-down GPU firmware: headless, no camera or video decode.
start_file=start_cd.elf
fixup_file=fixup_cd.dat
gpu_mem=16

kernel=Image
arm_64bit=1

disable_splash=1
boot_delay=0

# HiFiBerry MiniAmp (PCM5102A-style I2S DAC); onboard audio off.
dtparam=audio=off
dtoverlay=hifiberry-dac

# No Bluetooth: gives the full UART back to the serial console
# (GPIO14/15, ttyAMA0, 115200) and saves a little power.
dtoverlay=disable-bt
enable_uart=1

# Volume knob: a rotary encoder on GPIO17/27 (header pins 11/13) and its
# push switch on GPIO22 (pin 15), each to GND (pin 9 or 14). The kernel
# turns them into input events for the station (volume, play/pause). The
# internal pull-ups mean a bare encoder works; a module with its own
# pull-ups (KY-040 style) should have its + pin on 3.3 V, never 5 V.
gpio=17,27=ip,pu
dtoverlay=rotary-encoder,pin_a=17,pin_b=27,relative_axis=1
dtoverlay=gpio-key,gpio=22,active_low=1,gpio_pull=up,keycode=164,label="PLAYPAUSE"

# Real-time clock: a DS3231 (e.g. ZS-042) on I2C -- 3.3 V pin 1, SDA pin 3,
# SCL pin 5, GND pin 9. This also turns the I2C bus on. With no RTC fitted
# it's harmless. S12rtc sets the clock from it at boot; clock-save sets it
# from NTP.
dtoverlay=i2c-rtc,ds3231
