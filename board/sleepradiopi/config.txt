# SleepRadioPi boot config (Pi Zero 2 W). The boot partition is mounted
# read-only at /boot; edit this file on a PC if you need to change it.

# Cut-down GPU firmware: headless, no camera or video decode.
start_file=start_cd.elf
fixup_file=fixup_cd.dat
gpu_mem=16

kernel=Image
arm_64bit=1

disable_splash=1
boot_delay=0

# HiFiBerry MiniAmp (PCM5102A-style I2S DAC); onboard audio off.
dtparam=audio=off
dtoverlay=hifiberry-dac

# No Bluetooth: gives the full UART back to the serial console
# (GPIO14/15, ttyAMA0, 115200) and saves a little power.
dtoverlay=disable-bt
enable_uart=1

# Volume knob: a rotary encoder on GPIO17/27 (header pins 11/13) and its
# push switch on GPIO22 (pin 15), each to GND (pin 9 or 14). The kernel
# turns them into input events for the station (volume, play/pause). The
# internal pull-ups mean a bare encoder works; a module with its own
# pull-ups (KY-040 style) should have its + pin on 3.3 V, never 5 V.
gpio=17,27=ip,pu
dtoverlay=rotary-encoder,pin_a=17,pin_b=27,relative_axis=1
dtoverlay=gpio-key,gpio=22,active_low=1,gpio_pull=up,keycode=164,label="PLAYPAUSE"

# Preset buttons 1-4 (like a car radio's): momentary push buttons, each
# between its GPIO and GND -- 1: GPIO5 (pin 29), 2: GPIO6 (pin 31),
# 3: GPIO16 (pin 36), 4: GPIO8 (pin 24); GND pins 30, 34, 39 (or 25 / 20 for 4).
# They send KEY_1..KEY_4 to the station (io/presets.py). With none fitted: harmless.
# GPIO26 is the MiniAmp's SHUTDOWN line (button 4 there switched the amp off
# while held: a click), and GPIO16 its MUTE (button 3 mutes while it's held,
# silently, so it stayed). Keep other things off GPIO26.
dtoverlay=gpio-key,gpio=5,active_low=1,gpio_pull=up,keycode=2,label="PRESET1"
dtoverlay=gpio-key,gpio=6,active_low=1,gpio_pull=up,keycode=3,label="PRESET2"
dtoverlay=gpio-key,gpio=16,active_low=1,gpio_pull=up,keycode=4,label="PRESET3"
dtoverlay=gpio-key,gpio=8,active_low=1,gpio_pull=up,keycode=5,label="PRESET4"

# The cathedral radio (the Phonosphere, hardware "cathedral"): a 6-way
# rotary selector instead of the buttons -- its common to GND, positions 1-4
# on the preset pins above, 5 on GPIO23 (pin 16), 6 on GPIO24 (pin 18) --
# and a hidden back button on GPIO25 (pin 22) to GND (KEY_MENU: day/night,
# hold for the service menu). On the box radio: nothing fitted, harmless.
dtoverlay=gpio-key,gpio=23,active_low=1,gpio_pull=up,keycode=6,label="PRESET5"
dtoverlay=gpio-key,gpio=24,active_low=1,gpio_pull=up,keycode=7,label="PRESET6"
dtoverlay=gpio-key,gpio=25,active_low=1,gpio_pull=up,keycode=139,label="BACK"

# Hardware PWM for the cathedral's VU needle (GPIO12, pin 32, through ~5.6 k
# and a 2 k trimmer to the meter) and grille light (GPIO13, pin 33, to a
# logic-level MOSFET's gate). sleepradiopi-station exports both for the
# station (io/lamps.py). Nothing on those pins: harmless.
dtoverlay=pwm-2chan,pin=12,func=4,pin2=13,func2=4

# Real-time clock: a DS3231 (e.g. ZS-042) on I2C -- 3.3 V pin 1, SDA pin 3,
# SCL pin 5, GND pin 9. This also turns the I2C bus on. With no RTC fitted
# it's harmless. S12rtc sets the clock from it at boot; clock-save sets it
# from NTP.
dtoverlay=i2c-rtc,ds3231
